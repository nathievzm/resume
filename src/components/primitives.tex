% proportional token: bullet diameter scales with body size (~0.43em, radius ~2.02pt)
\def\BulletRadius{\fpeval{round(\TypeBase * 0.215, 2)}pt}

\NewDocumentCommand{\Bullet}{}{%
    \tikz[baseline=-0.50ex]{%
        \fill[InkMuted] (0,0) circle (\BulletRadius);%
    }%
}

\NewDocumentCommand{\DotSep}{}{%
    \hspace{\SpaceSix}%
    \tikz[baseline=-0.45ex]{%
        \fill[Hairline] (0,0) circle (1.45pt);%
    }%
    \hspace{\SpaceSix}%
}

% symmetrical em-dash separator (exact 4pt left and 4pt right)
\NewDocumentCommand{\DashSep}{}{%
    \hspace{4pt}{\TextMeta{—}}\hspace{4pt}%
}

\NewDocumentCommand{\SummaryBlock}{+m}{%
    {\raggedright\noindent\TextBody{#1}\par}%
}

% shared latex3 helpers (last-item flag check & zero-dependency orphan guard)
\ExplSyntaxOn
\NewDocumentCommand{\IfLastTF}{m m m}
  {
    \str_if_eq:nnTF { #1 } { last } { #2 } { #3 }
  }

\NewDocumentCommand{\NeedSpace}{m}
  {
    \par\penalty-100
    \dim_set:Nn \l_tmpa_dim { \pagegoal - \pagetotal }
    \dim_compare:nNnT { \l_tmpa_dim } < { #1 }
      {
        \dim_compare:nNnT { \l_tmpa_dim } > { 0pt }
          { \vfil\break }
      }
  }
\ExplSyntaxOff
