\newlength{\TimelineGutter}
\setlength{\TimelineGutter}{15pt}
\newsavebox{\RoleBox}

\NewDocumentCommand{\CompanyBar}{}{%
    \hspace*{2.75pt}%
    \tikz[baseline=-0.22ex]{%
        \fill[BrandPrimary, rounded corners=0.8pt] (0,-1.8pt) rectangle (2.5pt,7.8pt);%
    }%
    \hspace{5.5pt}%
}

\NewDocumentCommand{\CompanyGroup}{m m m}{%
    \NeedSpace{8\baselineskip}%
    \noindent%
    \CompanyBar%
    {\TextItemTitle{#1}}%
    \DashSep%
    {\TextMeta{#2}}%
    \hfill%
    {\TextDate{#3}}\par%
    \vspace{5.5pt}%
}

\NewDocumentEnvironment{TimelineRole}{O{continue} m m +b}{%
    \savebox{\RoleBox}{%
        \begin{minipage}[t]{\dimeval{\linewidth-\TimelineGutter}}%
            \raggedright\TextBody{#4}%
        \end{minipage}%
    }%
    \NeedSpace{\dimeval{\ht\RoleBox+\dp\RoleBox+2.5\baselineskip}}%
    \noindent%
    \makebox[\TimelineGutter][l]{%
        \hspace*{4pt}%
        \raisebox{0.45ex}{%
            \tikz[overlay]{%
                \IfLastTF{#1}{%
                    \shade[top color=BrandRule, bottom color=PaperBg]
                        (-0.675pt,0) rectangle (0.675pt,-\dimeval{\ht\RoleBox+\dp\RoleBox+4pt});%
                }{%
                    \draw[BrandRule, line width=1.35pt]
                        (0,0) -- (0,-\dimeval{\ht\RoleBox+\dp\RoleBox+28pt});%
                }%
                \fill[PaperBg] (0,0) circle (4.9pt);%
                \fill[BrandSoft] (0,0) circle (4.4pt);%
                \draw[BrandPrimary!55, line width=0.65pt] (0,0) circle (4.4pt);%
                \fill[BrandPrimary] (0,0) circle (2.1pt);%
            }%
        }%
    }%
    {\TextItemSubtitle{#2}}%
    \hfill%
    {\TextDate{#3}}\par%
    \vspace{3.2pt}%
    \noindent\hspace*{\TimelineGutter}\usebox{\RoleBox}\par%
    \IfLastTF{#1}{\vspace{14pt}}{\vspace{9pt}}%
}{}

\NewDocumentEnvironment{SingleRole}{O{continue} m m m m +b}{%
    \NeedSpace{6\baselineskip}%
    \noindent%
    \CompanyBar%
    {\TextItemTitle{#2}}%
    \DashSep%
    {\TextMeta{#3}}%
    \hfill%
    {\TextDate{#5}}\par%
    \vspace{4.5pt}%
    \noindent\hspace*{\TimelineGutter}{\TextItemSubtitle{#4}}\par%
    \vspace{3.2pt}%
    \noindent\hspace*{\TimelineGutter}\begin{minipage}[t]{\dimeval{\linewidth-\TimelineGutter}}{\raggedright\TextBody{#6}}\end{minipage}\par%
    \IfLastTF{#1}{}{\vspace{14pt}}%
}{}
