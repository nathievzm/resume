% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
% base font size and ratio for scaling
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

\def\TypeBase{10}
\def\TypeRatio{1.25}

% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
% font size steps
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

\NewDocumentCommand{\StepSize}{m}{%
    \fpeval{round(\TypeBase*\TypeRatio^#1, 1)}pt%
}

% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
% text styles
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

\NewDocumentCommand{\Heading}{m}{%
    {%
        \Quicksand%
        \WeightBold%
        \fontsize{\StepSize{4}}{\SpaceThirtySix}%
        \selectfont%
        \addfontfeature{LetterSpace=\TrackingTight}%
        \color{TextHeading}%
        #1%
    }%
}

\NewDocumentCommand{\TitleLarge}{m}{%
    {%
        \Quicksand%
        \WeightSemiBold%
        \fontsize{\StepSize{1}}{\SpaceTwenty}%
        \selectfont%
        \color{AccentSecondary}%
        #1%
    }%
}

\NewDocumentCommand{\TitleSmall}{m}{%
    {%
        \WeightSemiBold%
        \fontsize{\StepSize{0}}{\SpaceTwenty}%
        \selectfont%
        \color{TextHeading}%
        #1%
    }%
}

\NewDocumentCommand{\Body}{m}{%
    {%
        \WeightRegular%
        \fontsize{\StepSize{0}}{\SpaceTwenty}%
        \selectfont%
        \color{TextPrimary}%
        #1%
    }%
}

\NewDocumentCommand{\Accent}{m}{%
    {%
        \WeightSemiBold%
        \color{AccentSecondary}%
        #1%
    }%
}

\NewDocumentCommand{\LabelLarge}{m}{%
    {%
        \Quicksand%
        \WeightBold%
        \fontsize{\StepSize{0}}{\SpaceTwenty}%
        \selectfont%
        \addfontfeature{LetterSpace=\TrackingWidest}%
        \color{AccentPrimary}%
        \MakeUppercase{#1}%
    }%
}

\NewDocumentCommand{\LabelSmall}{m}{%
    {%
        \Quicksand%
        \WeightMedium%
        \fontsize{\StepSize{-1}}{\SpaceSixteen}%
        \selectfont%
        \color{TextSecondary}%
        #1%
    }%
}
