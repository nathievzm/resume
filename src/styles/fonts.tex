% ats note: Ligatures=TeX keeps fi/fl separate in the pdf text stream so
% parsers always read exact ascii keywords.

\setmainfont{Inter}[
    Path=../fonts/inter/,
    Extension=.ttf,
    Ligatures=TeX,
    UprightFont=*_18pt-Regular,
    BoldFont=*_18pt-Bold,
    ItalicFont=*_18pt-Italic,
    FontFace={mb}{n}{*_18pt-Medium}
]

\newfontfamily{\Sora}{Sora}[
    Path=../fonts/sora/,
    Extension=.ttf,
    Ligatures=TeX,
    UprightFont=*-SemiBold,
    BoldFont=*-Bold,
    FontFace={sb}{n}{*-SemiBold}
]

% github monaspace neon for dates, cli identifiers, and inline code
\setmonofont{MonaspaceNeon}[
    Path=../fonts/monaspace-neon/,
    Extension=.otf,
    Scale=0.84,
    UprightFont=*-Regular,
    BoldFont=*-Bold,
    FontFace={mb}{n}{*-Medium}
]
